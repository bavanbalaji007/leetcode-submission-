import java.util.*
 
fun main() {
    val sc = Scanner(System.`in`)
    val t = sc.nextInt()
    repeat(t) {
        val n = sc.nextInt()
        val freq = IntArray(n + 1)
        repeat(n) {
            val x = sc.nextInt()
            freq[x]++
        }
        var score = 0
        for (i in 1..n) {
            score += freq[i] / 2
        }
        println(score)
    }
}
